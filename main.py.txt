# Exo1
import numpy as np
from scipy.optimize import minimize

# Minimiser f(x,y) = (x−1)2 +(y −2)2 +xy
# Fonction à minimiser
def f_1(x):
    return (x[0] - 1)**2 + (x[1] - 2)**2 + x[0]*x[1]
# Point initial
x0_1 = np.array([0,0])
# Appel de minimize
result_1 = minimize(f_1, x0_1, method='BFGS')     
# Affichage des résultats
print("Exercice 1")
print("Solution␣optimale␣:", result_1.x)
print("Valeur␣de␣la␣fonction␣en␣minimum␣:", result_1.fun)

#exercice 2
#Minimiser C(x,y) = x2 +y2 − 3x − 4y +xy

def f_2(x):
    return x[0]**2 + x[1]**2 - 3*x[0] - 4*x[1] + x[0]*x[1]

print("Solution␣optimale␣:", result_2.x)

#OPTIMISATION AVEC CONTRAINTES
#exercice 3
#Minimiser f(x,y) = (x−2)2 +(y +1)2
#Contrainte␣: x +2y = 3 => x +2y − 3 = 0

def f_3(x):
    return (x[0] - 2)**2 + (x[1] + 1)**2

constraint_3 = {'type': 'eq', 'fun': lambda x: x[0] + 2*x[1] - 3}
x0_3 = np.array([0,0])
result_3 = minimize(f_3, x0_3, method='SLSQP', constraints=[constraint_3])
print("Exercice 3")
print("Solution␣optimale␣:", result_3.x)
print("Valeur␣de␣la␣fonction␣en␣minimum␣:", result_3.fun)

#exercice 4

#Contrainte␣: x +y = 100 => x +y − 100 = 0

def f_4(x):
